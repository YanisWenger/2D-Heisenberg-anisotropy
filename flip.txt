function SingleFlip(Lattice, i, j, T, L, σ, twopi, PBC)      # Propose a new configuration and either accept or reject it
    newspin = NewSpin(Lattice[:,i,j], σ, T)
    Δ = 0f0
    if PBC==false
        if i!=1
            Δ += cos(newspin[1]-Lattice[1,i-1,j])*cos(newspin[2]-Lattice[2,i-1,j]) - cos(Lattice[1,i,j]-Lattice[1,i-1,j])cos(Lattice[2,i,j]-Lattice[2,i-1,j])
        end
        if i != L
            Δ += cos(newspin[1]-Lattice[1,i+1,j])*cos(newspin[2]-Lattice[2,i+1,j]) - cos(Lattice[1,i,j]-Lattice[1,i+1,j])*cos(Lattice[2,i,j]-Lattice[2,i+1,j])
        end
        if j != 1
            Δ += cos(newspin[1]-Lattice[1,i,j-1])*cos(newspin[2]-Lattice[2,i,j-1]) - cos(Lattice[1,i,j]-Lattice[1,i,j-1])*cos(Lattice[2,i,j]-Lattice[2,i,j-1])
        end
        if j != L
            Δ += cos(newspin[1]-Lattice[1,i,j+1])*cos(newspin[2]-Lattice[2,i,j+1]) - cos(Lattice[1,i,j]-Lattice[1,i,j+1])*cos(Lattice[2,i,j]-Lattice[2,i,j+1])
        end
    else
        Δ += cos(newspin[1]-Lattice[1,mod(i-2,L)+1,j])*cos(newspin[2]-Lattice[2,mod(i-2,L)+1,j]) - cos(Lattice[1,i,j]-Lattice[1,mod(i-2,L)+1,j])*cos(Lattice[2,i,j]-Lattice[2,mod(i-2,L)+1,j])   +   cos(newspin[1]-Lattice[1,mod(i,L)+1,j])*cos(newspin[2]-Lattice[2,mod(i,L)+1,j]) - cos(Lattice[1,i,j]-Lattice[1,mod(i,L)+1,j])*cos(Lattice[2,i,j]-Lattice[2,mod(i,L)+1,j])   +   cos(newspin[1]-Lattice[1,i,mod(j-2,L)+1])*cos(newspin[2]-Lattice[2,i,mod(j-2,L)+1]) - cos(Lattice[1,i,j]-Lattice[1,i,mod(j-2,L)+1])*cos(Lattice[2,i,j]-Lattice[2,i,mod(j-2,L)+1])   +   cos(newspin[1]-Lattice[1,i,mod(j,L)+1])*cos(newspin[2]-Lattice[2,i,mod(j,L)+1]) - cos(Lattice[1,i,j]-Lattice[1,i,mod(j,L)+1])*cos(Lattice[2,i,j]-Lattice[2,i,mod(j,L)+1])
    end
    if exp(Δ/T) > rand(TaskLocalRNG(), Float32)
        Lattice[:,i,j] .= [mod(newspin[1], twopi), newspin[2]]
        acceptance=true
        E=-Δ
    else
        acceptance=false
        E=0
    end
    return Lattice[:,i,j], acceptance, E
end







function NewSpin(spin, σ, pi32)
    Which = rand(1:3)
    if Which == 1
        spin[1] = mod(rand(TaskLocalRNG(), Normal(spin[1], σ/sin(spin[2]))), 2*pi32)
    elseif Which == 2
        spin[2] = pi32 - spin[2]
    else
        spin[2] = mod(rand(TaskLocalRNG(), Normal(spin[2], σ)), 2*pi32)
        if spin[2] >= pi32
            spin[2] = 2*pi32-spin[2]
            spin[1] += pi32
        end
    end
    return spin, Which
end
